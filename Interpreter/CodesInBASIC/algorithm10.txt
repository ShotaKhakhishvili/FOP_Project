dim n as integer

print "Input an integer to find the corresponding fibonacci number"
input n

while n < 0
    print "Please enter a positive integer"
    input n
wend

if n < 3 then
    print "The answer: " n
end if

if n > 2 then
    dim prev as integer
dim fib as integer
    dim i as integer

    prev = 1
    fib = 1

    i = 2

    while i < n
        dim temp as integer
        temp = fib + prev
        prev = fib
        fib = temp
        i = i + 1
    wend

    print "The answer: " fib
end if
