dim n as integer
dim i as integer
dim answer as boolean

print "Input a number to find if it is prime or not"
input n

answer = true
i = 2

while i < n
    dim t as integer
    t = n % i
    if t = 0 then
        answer = false
    end if
    i = i + 1
wend

if answer = 1 then
    print "The answer is: prime"
end if

if answer = 0 then
    print "The answer is: not prime"
end if
